5TEST_CASE("Front panel switches are fixed and decode only I/O port bits", "[front_panel]") {
6 front_panel panel(0xFF, 0x10);
7 ram_card memory(0, 65536, 0x42);
8 bus cards;
9 cards.insert(&memory, 0);
10 cards.insert(&panel, 1);
11 REQUIRE(cards.read(0xFF) == 0x42);
12 REQUIRE(cards.read(0xFF, true) == 0x10);
13 REQUIRE(cards.read(0x12FF, true) == 0x10);
14 REQUIRE_FALSE(panel.in_range(0xFE));
15 cards.write(0xFF, 0xFF, true);
16 cards.write_force(0xFF, 0, true);
17 panel.raise_irq(true);
18 cards.clear();
19 REQUIRE(cards.read(0xFF, true) == 0x10);
20 REQUIRE_FALSE(panel.is_irq());
21 REQUIRE(panel.identify().name == "front panel");
22 REQUIRE(panel.identify().adr_range == 1);
