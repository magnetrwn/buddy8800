270class serial_card : public card {
271private:
272 const u16 start_adr;
273 const usize base_clock;
274 pty serial;
275 u8 control = 0;
276 u8 received = 0;
277 u8 transmitted = 0;
278 bool receive_full = false;
279 bool transmit_full = false;
280 std::string detail;
281
282 void refresh() {
283 if (transmit_full && serial.try_putch(transmitted))
284 transmit_full = false;
285 if (!receive_full)
286 receive_full = serial.try_getch(received);
287 }
288
289public:
290 serial_card(u16 start_adr, usize base_clock = SERIAL_BASE_CLOCK)
291 : start_adr(start_adr), base_clock(base_clock) {
292 if (start_adr > 0xFE)
293 throw std::out_of_range("Serial card requires two ports in 0..255");
294 serial.open();
295 }
296
297 const char* pty_name() const { return serial.name(); }
298 bool in_range(u16 adr) const override {
299 return (adr & 0xFF) >= start_adr && (adr & 0xFF) < start_adr + SERIAL_IO_ADDRESSES;
300 }
301 card_identify identify() override {
302 const usize divisor = (control & 3) == 1 ? 16 : (control & 3) == 2 ? 64 : 1;
303 detail = "baud: " + std::to_string(base_clock / divisor) +
304 ", ctrl: " + util::to_hex_s(static_cast<unsigned>(control), 2) +
305 ", pty: '" + serial.name() + "'";
306 return {start_adr, SERIAL_IO_ADDRESSES, "serial uart", detail.c_str()};
307 }
308 u8 read(u16 adr) override {
309 refresh();
310 if ((adr & 0xFF) == start_adr) {
311 // Polled 2SIO subset: receive ready and transmit buffer empty.
312 return (receive_full ? 0x01 : 0) | (transmit_full ? 0 : 0x02);
313 }
314 if ((adr & 0xFF) == start_adr + 1) {
315 const u8 byte = received;
316 receive_full = false;
317 return byte;
318 }
319 return BAD_U8;
320 }
321 void write(u16 adr, u8 byte) override {
322 refresh();
323 if ((adr & 0xFF) == start_adr) {
324 if ((byte & 3) == 3) { clear(); return; }
325 control = byte;
326 // Framing and clock selection are guest state. A PTY is a raw byte
327 // stream, and must not be reconfigured for every UART control write.
328 } else if ((adr & 0xFF) == start_adr + 1 && !transmit_full) {
329 transmitted = byte;
330 transmit_full = true;
331 refresh();
332 }
333 }
334 bool is_io() const override { return true; }
335 void clear() override {
336 control = received = transmitted = 0;
337 receive_full = transmit_full = false;
338 raise_irq(false);
339 }
340 void write_force(u16 adr, u8 byte) override { write(adr, byte); }
341 // Interrupt routing and physical modem/framing signals are not emulated.
342 std::array<u8, 3> get_irq() override { return {BAD_U8, BAD_U8, BAD_U8}; }
343};
