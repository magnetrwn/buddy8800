22 void setup(int argc, char** argv) {
23 if (argc < 1 or !(argc & 1))
24 throw std::invalid_argument("Invalid number of arguments. Provide pairs of ROM/data files and integer load addresses.");
25
26 processor.do_pseudo_bdos(conf.get_do_pseudo_bdos());
27 std::vector<u8> load_rom_vec;
28
29 // The arguments come in pairs of filename and location to load the ROM at.
30 for (int i = 1; i < argc; i += 2) {
31 std::ifstream load_rom(argv[i], std::ios::binary);
32
33 if (!load_rom)
34 throw std::runtime_error("Could not open file: " + std::string(argv[i]));
35
36 load_rom_vec.assign(std::istreambuf_iterator<char>(load_rom), {});
37
38 // The first ROM is the one that will have the reset vector jump to.
39 const std::string address(argv[i + 1]);
40 usize consumed = 0;
41 const auto offset = std::stoul(address, &consumed, 0);
42 if (consumed != address.size() || offset > 65535 || address.front() == '-')
43 throw std::invalid_argument("Invalid load address: " + address);
44 if (load_rom.bad() || load_rom_vec.empty())
45 throw std::runtime_error("Empty or unreadable ROM file");
46 processor.load(load_rom_vec.begin(), load_rom_vec.end(), offset, i == 1 && offset > 2);
47 }
48
49 processor.set_pc(conf.get_start_pc());
50 }
51
52 void run(const volatile std::sig_atomic_t& stop) {
53 while (!stop && !processor.is_halted()) {
54 processor.step();
55 if (cardbus.is_irq())
56 processor.interrupt(cardbus.get_irq());
57 }
58 }
59
60 std::string info() const { return cardbus.bus_map_s(); }

62 emulator(const char* config_filename)
63 : conf(config_filename),
64 cardbus(conf.get_bus()),
65 processor(cardbus, conf.get_start_pc() == 0x0000) {}
