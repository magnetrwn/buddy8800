73 static std::string get_absolute_dir() {

89 constexpr static bool get_parity(T value) {
90 static_assert(std::is_integral<T>::value, "T must be an integral type");
91 usize p = 0;
92 for (usize i = 0; i < 8 * sizeof(T); ++i)
93 p += (value >> i) & 1;
