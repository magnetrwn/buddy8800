103 // An optional observer supplies instruction records to the frontend.
104 // No terminal output or ncurses dependencies belong in the interpreter.
105 std::function<void(u16, const std::array<u8, 3>&, usize)> trace_observer;
106
107 template <usize ops>
108 void trace_instruction(u8 opcode) {
109 if (!trace_observer)
110 return;
111 std::array<u8, 3> bytes{opcode, 0, 0};
112 for (usize i = 1; i < ops; ++i)
113 bytes[i] = fetch_ptr == &cpu::fetch_ext
114 ? ext_op[i - 1]
115 : static_cast<u8>(cardbus[static_cast<u16>(state.PC() + i - 1)]);
116 trace_observer(static_cast<u16>(state.PC() - 1), bytes, ops);
117 }

119 bool resolve_flag_cond(u8 cc) {
120 switch (cc) {
121 case 0b000:
122 return !state.get_flag(cpu_flags::Z);
123 case 0b001:
124 return state.get_flag(cpu_flags::Z);
125 case 0b010:
126 return !state.get_flag(cpu_flags::C);
127 case 0b011:
128 return state.get_flag(cpu_flags::C);
129 case 0b100:
130 return !state.get_flag(cpu_flags::P);
131 case 0b101:
132 return state.get_flag(cpu_flags::P);
133 case 0b110:
134 return !state.get_flag(cpu_flags::S);
135 case 0b111:
136 return state.get_flag(cpu_flags::S);
137 default:
138 return false;
139 }
140 }

142 void handle_bdos() {
143 if (state.PC() == 0x0000) {
144 if (allow_reset_twice) {
145
146 allow_reset_twice = false;
147 return;
148 }
149
150 cardbus[0] = 0b01110110;
151 }
152
153 if (state.PC() == 0x0005) {
154 u8 c = state.C();
155
156 if (c == 0x02)
157 printer << state.E();
158
159 else if (c == 0x09)
160 for (u16 de = state.DE(); cardbus[de] != '$'; ++de)
161 printer << cardbus[de];
162
163 else
164 throw std::runtime_error("Unknown BDOS 0x0005 call parameters.");
165
166 fetch();
167
168 RETURN();
169 }
170 }

303 inline void alu_impl(cpu_registers8 src, u8 operation, bool is_immediate) {
304 u16 a = state.A();
305 u16 with =
306 (is_immediate) ? fetch() : ((is_memref(src)) ? cardbus[state.HL()] : state.get_register8(src));
307 u16 result;
308
309 switch (operation) {
310 case 0b000:
311 case 0b001:
312 result = a + with + ((operation == 0b001) ? state.flgC() : 0);
313 state.flgAC(((a & 0x0F) + (with & 0x0F) + ((operation == 0b001) ? state.flgC() : 0)) > 0x0F);
314 break;
315 case 0b010:
316 case 0b011:
317 result = a - with - ((operation == 0b011) ? state.flgC() : 0);
318 state.flgAC((a & 0x0F) >= (with & 0x0F));
319 break;
320 case 0b100:
321 result = a & with;
322 state.flgAC((a | with) & 0x08);
323 break;
324 case 0b101:
325 case 0b110:
326 result = (operation == 0b101) ? (a ^ with) : (a | with);
327 state.flgAC(false);
328 break;
329 case 0b111:
330 result = a - with;
331 state.flgAC((a & 0x0F) >= (with & 0x0F));
332 state.flgC(result & 0x100);
333 state.set_Z_S_P_flags(result);
334 break;
335 default:
336 throw std::runtime_error("Invalid ALU operation.");
337 }
338
339 state.flgC(result & 0x100);
340 state.set_Z_S_P_flags(result);
341 if (operation != 0b111)
342 state.A(result);
343 }
