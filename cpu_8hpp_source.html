88 template <usize ops>
89 void _trace(u8 opcode) {
90#ifndef DISABLE_TRACE
91 if (!trace_observer) return;
92 std::array<u8, 3> bytes{opcode, 0, 0};
93 for (usize i = 1; i < ops; ++i)
94 bytes[i] = fetch_ptr == &cpu::fetch_ext
95 ? ext_op[i - 1] : static_cast<u8>(cardbus[static_cast<u16>(state.PC() + i - 1)]);
96 trace_observer(static_cast<u16>(state.PC() - 1), bytes, ops);
97#else
98 (void)opcode;
99#endif
100 }

102 bool resolve_flag_cond(u8 cc) {
103 switch (cc) {
104 case 0b000: return !state.get_flag(cpu_flags::Z);
105 case 0b001: return state.get_flag(cpu_flags::Z);
106 case 0b010: return !state.get_flag(cpu_flags::C);
107 case 0b011: return state.get_flag(cpu_flags::C);
108 case 0b100: return !state.get_flag(cpu_flags::P);
109 case 0b101: return state.get_flag(cpu_flags::P);
110 case 0b110: return !state.get_flag(cpu_flags::S);
111 case 0b111: return state.get_flag(cpu_flags::S);
112 default: return false;
113 }

259 inline void _ALU(cpu_registers8 src, u8 alu, bool is_immediate) {
260 u16 a = state.A();
261 u16 with = (is_immediate) ? fetch() : ((is_memref(src)) ? cardbus[state.HL()] : state.get_register8(src));
262 u16 result;
263
264 switch (alu) {
265 case 0b000:
266 case 0b001:
267 result = a + with + ((alu == 0b001) ? state.flgC() : 0);
268 state.flgAC(((a & 0x0F) + (with & 0x0F) + ((alu == 0b001) ? state.flgC() : 0)) > 0x0F);
269 break;
270 case 0b010:
271 case 0b011:
272 result = a - with - ((alu == 0b011) ? state.flgC() : 0);
273 state.flgAC((a & 0x0F) >= (with & 0x0F));
274 break;
275 case 0b100:
276 result = a & with;
277 state.flgAC((a | with) & 0x08);
278 break;
279 case 0b101:
280 case 0b110:
281 result = (alu == 0b101) ? (a ^ with) : (a | with);
282 state.flgAC(false);
283 break;
284 case 0b111:
285 result = a - with;
286 state.flgAC((a & 0x0F) >= (with & 0x0F));
287 state.flgC(result & 0x100);
288 state.set_Z_S_P_flags(result);
289 break;
290 default: throw std::runtime_error("Invalid ALU operation.");
291 }
292
293 state.flgC(result & 0x100);
294 state.set_Z_S_P_flags(result);
295 if (alu != 0b111)
296 state.A(result);
297 }

811 void load(T begin, T end, usize offset = 0, bool auto_reset_vector = false) {
812 usize dist = std::distance(begin, end);
813
814 if (dist > cardbus.size() - offset)
815 throw std::out_of_range("Not enough space in emulated cardbus.");
816
817 if constexpr (std::is_same_v<bus_iface, bus&>)
818 for (usize i = 0; i < dist; ++i)
819 cardbus.write_force(offset + i, *(begin + i));
820 else
821 for (usize i = 0; i < dist; ++i)
822 cardbus[offset + i] = *(begin + i);
823
824 if (auto_reset_vector) {
825 if (offset <= 2)
826 throw std::out_of_range("First program bytes will be overwritten by reset vector.");
827
828 cardbus[0] = 0xC3;
829 cardbus[1] = static_cast<u8>(offset & 0xFF);
830 cardbus[2] = static_cast<u8>(offset >> 8);
831 }
